모바일 웹 환경에서 **상단 탭 버튼 클릭** 및 **좌우 스와이프**를 통해 부드럽게 전환되고, **내부 세로 스크롤과 간섭하지 않는** 단일 파일(`index.html`) 코드입니다.

CSS의 **Scroll Snap**과 모바일 브라우저의 **네이티브 터치 방향 제어(Direction Lock)** 기능을 활용하여 자바스크립트 터치 이벤트 연산으로 인한 딜레이 없이 60fps의 매끄러운 터치감을 제공합니다. 또한 스와이프 움직임에 맞춰 **상단 인디케이터 바가 실시간으로 연동되어 이동**합니다.

  <script>
    const container = document.getElementById('swipeContainer');
    const indicator = document.getElementById('tabIndicator');
    const tabBtns = document.querySelectorAll('.tab-btn');
    const pageCount = tabBtns.length;

    let isProgrammaticScroll = false; // 버튼 클릭으로 인한 스크롤 중인지 여부

    // 1. 실시간 스크롤 감지 및 인디케이터 바 위치 계산
    container.addEventListener('scroll', () => {
      const scrollLeft = container.scrollLeft;
      const width = container.clientWidth;
      
      // 현재 진행률 (0.0 ~ 2.0)
      const progress = scrollLeft / width;
      
      // 인디케이터 바를 진행률만큼 실시간 변환(TranslateX)
      // 탭 1개당 너비가 100% (전체 인디케이터 바 너비 기준)이므로 progress * 100% 이동
      indicator.style.transform = `translateX(${progress * 100}%)`;

      // 스와이프 상태에서 활성화된 탭 버튼 하이라이트 제어
      if (!isProgrammaticScroll) {
        const activeIndex = Math.round(progress);
        updateActiveTabUI(activeIndex);
      }
    });

    // 2. 탭 버튼 클릭 시 스크롤 이동
    tabBtns.forEach((btn) => {
      btn.addEventListener('click', (e) => {
        const targetIndex = parseInt(e.currentTarget.getAttribute('data-index'));
        
        isProgrammaticScroll = true;
        updateActiveTabUI(targetIndex);

        // 해당 위치로 부드럽게 스크롤
        container.scrollTo({
          left: targetIndex * container.clientWidth,
          behavior: 'smooth'
        });

        // 애니메이션 완료 후 스크롤 상태 해제
        setTimeout(() => {
          isProgrammaticScroll = false;
        }, 350);
      });
    });

    // 활성화 탭 Class 업데이트 함수
    function updateActiveTabUI(index) {
      tabBtns.forEach((btn, idx) => {
        if (idx === index) {
          btn.classList.add('active');
        } else {
          btn.classList.remove('active');
        }
      });
    }

    // 창 크기 변경 시 스크롤 위치 보정
    window.addEventListener('resize', () => {
      const activeBtn = document.querySelector('.tab-btn.active');
      const index = parseInt(activeBtn.getAttribute('data-index'));
      container.scrollLeft = index * container.clientWidth;
    });
  </script>
