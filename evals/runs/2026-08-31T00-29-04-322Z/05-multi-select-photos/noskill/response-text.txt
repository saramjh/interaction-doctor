문제의 원인은 `click` 이벤트 리스너 안에서 눌린 사진(`tile`)이 아니라, 선택 모드에 들어갈 때 저장해둔 첫 번째 사진(`activeTile`)을 고정으로 전달하고 있었기 때문입니다.

또한, 롱프레스(길게 누르기)가 끝나고 손을 뗄 때 브라우저에서 `click` 이벤트가 연달아 발생하여 방금 선택된 첫 번째 사진이 곧바로 다시 해제되는 현상도 방지할 수 있도록 보완했습니다.

### 주요 수정 사항
1. **`activeTile` 참조 오류 수정**: `click` 이벤트 시 `activeTile` 대신 현재 클릭한 요소(`tile`, `i`)를 `toggleSelection`에 전달하도록 변경했습니다.
2. **롱프레스 직후 클릭 이벤트 방지**: 롱프레스로 선택 모드에 들어간 직후 손을 뗄 때 발생되는 클릭 이벤트로 인해 첫 번째 사진이 도로 해제되는 현상을 `isLongPress` 플래그로 방지했습니다.

<script>
// 사진을 길게 누르면 다중 선택 모드로 들어간다.
// 다중 선택 모드에서는 다른 사진을 탭하면 그 사진도 선택 목록에 추가된다.

const PHOTO_COUNT = 9;
const gridEl = document.getElementById("grid");
const headerEl = document.getElementById("header");

let selectMode = false;
const selected = new Set();

function updateHeader() {
	headerEl.textContent = selectMode ? selected.size + "장 선택됨" : "사진 " + PHOTO_COUNT + "장";
}

function enterSelectMode(tile, index) {
	selectMode = true;
	selected.add(index);
	tile.classList.add("selected");
	updateHeader();
}

function toggleSelection(tile, index) {
	if (selected.has(index)) {
		selected.delete(index);
		tile.classList.remove("selected");
	} else {
		selected.add(index);
		tile.classList.add("selected");
	}
	updateHeader();
}

for (let i = 0; i < PHOTO_COUNT; i++) {
	const tile = document.createElement("div");
	tile.className = "photo";
	tile.dataset.index = String(i);
	tile.innerHTML = '<div class="checkmark">✓</div>';
	gridEl.appendChild(tile);

	let pressTimer = null;
	let isLongPress = false;

	tile.addEventListener("pointerdown", () => {
		isLongPress = false;
		pressTimer = setTimeout(() => {
			isLongPress = true;
			if (!selectMode) enterSelectMode(tile, i);
		}, 500);
	});

	tile.addEventListener("pointerup", () => {
		clearTimeout(pressTimer);
	});

	tile.addEventListener("pointercancel", () => {
		clearTimeout(pressTimer);
	});

	// 다중 선택 모드에서 사진을 탭하면 선택/해제를 토글한다.
	tile.addEventListener("click", () => {
		if (isLongPress) {
			isLongPress = false;
			return; // 롱프레스로 선택 모드 진입 직후의 클릭 감지 무시
		}
		if (!selectMode) return;
		toggleSelection(tile, i);
	});
}
</script>
</body>
</html>
