모바일 계약서에 최적화된 **전자 서명 캔버스 컴포넌트**입니다. 

터치/펜 조작 시 발생할 수 있는 화면 스크롤 및 바운스(Rubber-banding) 현상을 완전히 방지하도록 `touch-action: none`과 이벤트 제어 로직을 적용하였으며, 고해상도(Retina/High-DPI) 모바일 디스플레이에서도 부드럽고 선명한 서명이 가능하도록 구현되었습니다.

    <script>
        document.addEventListener('DOMContentLoaded', () => {
            const canvas = document.getElementById('signatureCanvas');
            const container = document.getElementById('canvasContainer');
            const guideText = document.getElementById('guideText');
            const btnClear = document.getElementById('btnClear');
            const btnSubmit = document.getElementById('btnSubmit');
            const modal = document.getElementById('resultModal');
            const signatureImage = document.getElementById('signatureImage');
            const btnCloseModal = document.getElementById('btnCloseModal');

            const ctx = canvas.getContext('2d');
            
            let isDrawing = false;
            let hasSigned = false;
            let lastX = 0;
            let lastY = 0;

            // 해상도 조절 (고해상도 디스플레이 선명도 유지)
            function resizeCanvas() {
                const ratio = Math.max(window.devicePixelRatio || 1, 1);
                
                // 엘리먼트의 실제 픽셀 크기 가져오기
                const rect = container.getBoundingClientRect();
                
                canvas.width = rect.width * ratio;
                canvas.height = rect.height * ratio;
                
                ctx.scale(ratio, ratio);
                
                // 리사이즈 후 스타일 재설정
                resetContextStyle();
            }

            function resetContextStyle() {
                ctx.strokeStyle = '#0f172a'; // 서명 펜 색상
                ctx.lineWidth = 3;            // 서명 펜 두께
                ctx.lineCap = 'round';        // 선 끝 모양을 둥글게
                ctx.lineJoin = 'round';       // 선 연결부를 둥글게
            }

            // 좌표 계산 함수 (스크롤 및 오프셋 보정)
            function getCoordinates(event) {
                const rect = canvas.getBoundingClientRect();
                let clientX = event.clientX;
                let clientY = event.clientY;

                // TouchEvent 처리
                if (event.touches && event.touches.length > 0) {
                    clientX = event.touches[0].clientX;
                    clientY = event.touches[0].clientY;
                }

                return {
                    x: clientX - rect.left,
                    y: clientY - rect.top
                };
            }

            // 드로잉 시작
            function startDrawing(e) {
                isDrawing = true;
                const pos = getCoordinates(e);
                lastX = pos.x;
                lastY = pos.y;

                // 점 찍기 대응 (터치만 하고 떼었을 때)
                ctx.beginPath();
                ctx.arc(lastX, lastY, ctx.lineWidth / 2, 0, Math.PI * 2);
                ctx.fillStyle = ctx.strokeStyle;
                ctx.fill();

                if (!hasSigned) {
                    hasSigned = true;
                    guideText.style.opacity = '0';
                    container.classList.add('active');
                }
            }

            // 드로잉 진행 (베지에 곡선을 통한 부드러운 선 표현)
            function draw(e) {
                if (!isDrawing) return;

                // iOS/안드로이드 스크롤 및 바운스 강제 방지
                if (e.cancelable) {
                    e.preventDefault();
                }

                const pos = getCoordinates(e);
                
                ctx.beginPath();
                ctx.moveTo(lastX, lastY);

                // 부드러운 선을 위한 중점 계산 (Quadratic Curve)
                const midX = (lastX + pos.x) / 2;
                const midY = (lastY + pos.y) / 2;
                ctx.quadraticCurveTo(lastX, lastY, midX, midY);
                ctx.lineTo(pos.x, pos.y);
                
                ctx.stroke();

                lastX = pos.x;
                lastY = pos.y;
            }

            // 드로잉 종료
            function stopDrawing() {
                if (isDrawing) {
                    isDrawing = false;
                    ctx.beginPath();
                }
            }

            // 캔버스 초기화 (지우기)
            function clearCanvas() {
                ctx.clearRect(0, 0, canvas.width, canvas.height);
                hasSigned = false;
                guideText.style.opacity = '1';
                container.classList.remove('active');
            }

            // --- 이벤트 리스너 등록 ---

            // Pointer Events (최신 브라우저 통합: 마우스, 터치, 스탈러스 펜)
            canvas.addEventListener('pointerdown', (e) => {
                canvas.setPointerCapture(e.pointerId);
                startDrawing(e);
            });

            canvas.addEventListener('pointermove', draw);

            canvas.addEventListener('pointerup', (e) => {
                canvas.releasePointerCapture(e.pointerId);
                stopDrawing();
            });

            canvas.addEventListener('pointercancel', (e) => {
                canvas.releasePointerCapture(e.pointerId);
                stopDrawing();
            });

            // 화면 스크롤 방지를 위한 Touch 이벤트 PreventDefault 보완
            canvas.addEventListener('touchstart', (e) => {
                if (e.cancelable) e.preventDefault();
            }, { passive: false });

            canvas.addEventListener('touchmove', (e) => {
                if (e.cancelable) e.preventDefault();
            }, { passive: false });

            // 초기화 버튼 클릭
            btnClear.addEventListener('click', clearCanvas);

            // 서명 완료 버튼 클릭 (Base64 이미지 추출 예시)
            btnSubmit.addEventListener('click', () => {
                if (!hasSigned) {
                    alert('서명을 작성해 주세요.');
                    return;
                }

                // 캔버스 내용을 PNG Base64 DataURL로 변환
                const dataURL = canvas.toDataURL('image/png');
                signatureImage.src = dataURL;
                modal.classList.add('open');
            });

            // 모달 닫기
            btnCloseModal.addEventListener('click', () => {
                modal.classList.remove('open');
            });

            // 창 크기 변경 시 대응
            window.addEventListener('resize', resizeCanvas);

            // 최초 캔버스 사이즈 설정
            resizeCanvas();
        });
    </script>
</body>
</html>
