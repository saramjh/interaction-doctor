모바일 브라우저에서 화면 스크롤 대신 곡 순서 변경(드래그)이 정상 동작하도록 수정했습니다.

### 주요 수정 내용
1. **CSS `touch-action: none;` 추가**: 터치 디바이스에서 트랙 항목을 터치한 채 움직일 때 브라우저의 기본 화면 스크롤 동작이 발동하지 않도록 설정했습니다.
2. **`setPointerCapture` 적용**: 포인터 Down 시 해당 이벤트를 요소에 고정(Capture)하여, 터치가 이동하거나 요소를 벗어나도 포인터 Move/Up 이벤트가 안정적으로 수신되도록 수정했습니다.
3. **`pointercancel` 예외 처리**: 드래그 도중 시스템 팝업 등 외부 이벤트로 터치가 중단되었을 때 상태를 원복하도록 예외 처리를 추가했습니다.

수정된 전체 파일 내용입니다:

```html
<!doctype html>
<html lang="ko">
<head>
<meta charset="utf-8" />
<meta name="viewport" content="width=device-width, initial-scale=1" />
<title>내 재생목록</title>
<style>
	* { box-sizing: border-box; }
	html, body {
		margin: 0;
		height: 100%;
		font-family: -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, sans-serif;
		background: #0f0f13;
		color: #eee;
	}
	header {
		padding: 16px;
		font-size: 18px;
		font-weight: 700;
		border-bottom: 1px solid #26262e;
	}
	#playlist {
		overflow-y: auto;
		height: calc(100% - 56px);
		padding: 8px;
	}
	.track {
		display: flex;
		align-items: center;
		gap: 12px;
		padding: 12px;
		margin-bottom: 6px;
		background: #1b1b22;
		border-radius: 10px;
		user-select: none;
		touch-action: none;
		position: relative;
	}
	.track.dragging {
		background: #2a2a36;
		z-index: 10;
	}
	.cover {
		width: 44px;
		height: 44px;
		border-radius: 6px;
		background: linear-gradient(135deg, #6a5acd, #ff6b9a);
		flex-shrink: 0;
	}
	.meta { flex: 1; min-width: 0; }
	.title { font-size: 14px; font-weight: 600; white-space: nowrap; overflow: hidden; text-overflow: ellipsis; }
	.artist { font-size: 12px; color: #9a9aa5; }
	.duration { font-size: 12px; color: #9a9aa5; }
</style>

<body>
<header>내 재생목록 · 24곡</header>
<div id="playlist"></div>

<script>
// 재생목록 순서를 손가락으로 눌러서 드래그하면 바꿀 수 있어야 한다.
// (핸들 아이콘 없음 — 항목 전체가 드래그 타겟)

const TRACKS = [
	["Midnight City", "M83", "4:03"],
	["Take On Me", "a-ha", "3:46"],
	["Circles", "Post Malone", "3:35"],
	["Blinding Lights", "The Weeknd", "3:20"],
	["Levitating", "Dua Lipa", "3:23"],
	["As It Was", "Harry Styles", "2:47"],
	["Heat Waves", "Glass Animals", "3:58"],
	["Stay", "The Kid LAROI", "2:21"],
	["good 4 u", "Olivia Rodrigo", "2:58"],
	["Watermelon Sugar", "Harry Styles", "2:54"],
];

const listEl = document.getElementById("playlist");

function render() {
	listEl.innerHTML = "";
	TRACKS.forEach((t, i) => {
		const row = document.createElement("div");
		row.className = "track";
		row.dataset.index = String(i);
		row.innerHTML =
			'<div class="cover"></div>' +
			'<div class="meta"><div class="title">' + t[0] + '</div><div class="artist">' + t[1] + '</div></div>' +
			'<div class="duration">' + t[2] + '</div>';
		listEl.appendChild(row);
		attachDrag(row);
	});
}

// 드래그 구현: 항목을 누른 채로 옮기면 그 위치로 순서가 바뀐다.
function attachDrag(row) {
	let startY = 0;
	let isDragging = false;

	row.addEventListener("pointerdown", (e) => {
		isDragging = true;
		startY = e.clientY;
		row.classList.add("dragging");
		row.setPointerCapture(e.pointerId);
	});

	row.addEventListener("pointermove", (e) => {
		if (!isDragging) return;
		const dy = e.clientY - startY;
		row.style.transform = "translateY(" + dy + "px)";
	});

	row.addEventListener("pointerup", (e) => {
		if (!isDragging) return;
		isDragging = false;
		row.classList.remove("dragging");
		row.style.transform = "";

		try {
			row.releasePointerCapture(e.pointerId);
		} catch (err) {}

		const dy = e.clientY - startY;
		const idx = Number(row.dataset.index);
		const rowHeight = row.getBoundingClientRect().height + 6;
		const shift = Math.round(dy / rowHeight);
		const newIdx = Math.min(TRACKS.length - 1, Math.max(0, idx + shift));
		if (newIdx !== idx) {
			const [moved] = TRACKS.splice(idx, 1);
			TRACKS.splice(newIdx, 0, moved);
			render();
		}
	});

	row.addEventListener("pointercancel", () => {
		if (!isDragging) return;
		isDragging = false;
		row.classList.remove("dragging");
		row.style.transform = "";
	});
}

render();
</script>
</body>
</html>
```
