발생하고 있는 문제의 원인은 포인터 이벤트(Pointer Events)의 **취소 처리 미비**와 **멀티 터치(다중 포인터) 제어 부족** 때문입니다.

### 1. 손도 안 댔는데 갑자기 항목이 옮겨지거나 메뉴가 뜨는 이유
* 모바일 환경이나 터치스크린에서 화면을 스크롤하거나 다른 제스처를 취할 때 브라우저는 `pointercancel` 이벤트를 발생시킵니다.
* 기존 코드에는 `pointercancel` 이벤트 리스너가 없어, 터치가 취소되었음에도 `setTimeout` 타이머가 계속 돌아가 500ms/1000ms 뒤에 메뉴가 뜨거나 항목이 드래그(투명도 변경) 모드로 바뀌어 버렸던 것입니다.

### 2. 두 손가락으로 누르면 메뉴가 두 번 뜨는 이유
* 각 연락처 항목이 별도로 `pointerdown`을 수신하므로, 두 손가락으로 서로 다른 항목을 터치하면 두 곳 모두에서 롱프레스 타이머가 동작합니다.
* 주 포인터(첫 번째 입력) 여부(`e.isPrimary`)를 체크하지 않아 발생한 현상입니다.

---

### 수정 내용
1. **`pointercancel` 이벤트 처리 추가**: 스크롤되거나 터치가 중단될 때 타이머 및 드래그 상태를 즉시 취소하도록 설정했습니다.
2. **`e.isPrimary` 및 포인터 ID 추적**: 주 포인터(첫 번째 손가락 또는 마우스)만 이벤트를 처리하고, 다중 터치를 차단했습니다.
3. **`setPointerCapture` 적용**: 포인터가 요소를 약간 벗어나더라도 이벤트를 안정적으로 추적하고 릴리스하도록 개선했습니다.

수정된 전체 `index.html` 파일 코드입니다.

```html
<!doctype html>
<html lang="ko">
<head>
<meta charset="utf-8" />
<meta name="viewport" content="width=device-width, initial-scale=1" />
<title>즐겨찾는 연락처</title>
<style>
	* { box-sizing: border-box; }
	html, body {
		margin: 0;
		height: 100%;
		font-family: -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, sans-serif;
		background: #f5f5f7;
		color: #1c1c1e;
	}
	header {
		padding: 16px;
		font-size: 18px;
		font-weight: 700;
		background: #fff;
		border-bottom: 1px solid #e2e2e6;
	}
	#list { padding: 12px; }
	.contact {
		display: flex;
		align-items: center;
		gap: 12px;
		background: #fff;
		border-radius: 10px;
		padding: 12px 14px;
		margin-bottom: 8px;
		user-select: none;
		position: relative;
		touch-action: pan-y;
	}
	.contact.selected { outline: 2px solid #007aff; }
	.avatar {
		width: 40px;
		height: 40px;
		border-radius: 50%;
		background: linear-gradient(135deg, #667eea, #764ba2);
		flex-shrink: 0;
	}
	.name { font-size: 15px; font-weight: 600; }
	#context-menu {
		position: fixed;
		background: #2c2c2c;
		color: #fff;
		border-radius: 10px;
		overflow: hidden;
		display: none;
		z-index: 100;
		min-width: 130px;
	}
	#context-menu button {
		display: block;
		width: 100%;
		padding: 12px 16px;
		background: none;
		border: none;
		color: inherit;
		font-size: 14px;
		text-align: left;
	}
</style>
</head>
<body>
<header id="header">즐겨찾는 연락처</header>
<div id="list"></div>
<div id="context-menu">
	<button>전화</button>
	<button>메시지</button>
	<button>삭제</button>
</div>

<script>
// 즐겨찾는 연락처를 길게 누르면:
//   - 그대로 누르고 있으면(움직이지 않으면) 500ms에 메뉴(전화/메시지/삭제)가 뜬다.
//   - 메뉴가 뜬 뒤 계속 누르고 있으면 1000ms에 여러 개 고르는 모드로 바뀐다.
//   - 누른 채로 위아래로 옮기면(8px 이상) 그 자리에서 순서가 바뀐다.

const CONTACTS = ["김민준", "이서연", "박도윤", "최하늘", "정예린"];
const listEl = document.getElementById("list");
const headerEl = document.getElementById("header");
const menuEl = document.getElementById("context-menu");

let selectMode = false;
const selected = new Set();

function updateHeader() {
	headerEl.textContent = selectMode ? selected.size + "명 선택됨" : "즐겨찾는 연락처";
}

function render() {
	listEl.innerHTML = "";
	CONTACTS.forEach((name, i) => {
		const row = document.createElement("div");
		row.className = "contact";
		row.dataset.index = String(i);
		row.innerHTML = '<div class="avatar"></div><div class="name">' + name + "</div>";
		listEl.appendChild(row);
		attachTripleGesture(row, i);
	});
}

function openContextMenu(x, y) {
	menuEl.style.left = x + "px";
	menuEl.style.top = y + "px";
	menuEl.style.display = "block";
}

function enterSelectMode(row, index) {
	selectMode = true;
	selected.add(index);
	row.classList.add("selected");
	updateHeader();
}

// 롱프레스 3중 분기: 이동하면 재정렬, 안 움직이면 500ms에 메뉴,
// 메뉴 뜨고도 계속 누르고 있으면 1000ms에 다중 선택.
function attachTripleGesture(row, index) {
	let startX = 0;
	let startY = 0;
	let moved = false;
	let contextMenuTimer = null;
	let escalateTimer = null;
	let contextMenuFired = false;
	let activePointerId = null;

	const clearTimersAndState = () => {
		clearTimeout(contextMenuTimer);
		clearTimeout(escalateTimer);
		contextMenuTimer = null;
		escalateTimer = null;
		row.style.opacity = "";
		if (activePointerId !== null) {
			try {
				row.releasePointerCapture(activePointerId);
			} catch (e) {}
			activePointerId = null;
		}
	};

	row.addEventListener("pointerdown", (e) => {
		// 주 포인터(첫 손가락/마우스)만 처리하고 다중 터치는 무시
		if (!e.isPrimary) return;
		if (activePointerId !== null) return;

		activePointerId = e.pointerId;
		try {
			row.setPointerCapture(e.pointerId);
		} catch (err) {}

		startX = e.clientX;
		startY = e.clientY;
		moved = false;
		contextMenuFired = false;

		contextMenuTimer = setTimeout(() => {
			if (!moved) {
				contextMenuFired = true;
				openContextMenu(e.clientX, e.clientY);
				escalateTimer = setTimeout(() => {
					if (!moved) enterSelectMode(row, index);
				}, 500);
			}
		}, 500);
	});

	row.addEventListener("pointermove", (e) => {
		if (e.pointerId !== activePointerId) return;
		if (moved) return;
		const dx = e.clientX - startX;
		const dy = e.clientY - startY;
		if (Math.hypot(dx, dy) > 8) {
			moved = true;
			clearTimeout(contextMenuTimer);
			clearTimeout(escalateTimer);
			if (!contextMenuFired) {
				row.style.opacity = "0.6"; // 재정렬 시작 표시
			}
		}
	});

	row.addEventListener("pointerup", (e) => {
		if (e.pointerId !== activePointerId) return;
		clearTimersAndState();
	});

	// 스크롤 발생이나 터치 취소 시 타이머를 확실히 해제
	row.addEventListener("pointercancel", (e) => {
		if (e.pointerId !== activePointerId) return;
		clearTimersAndState();
	});
}

document.addEventListener("click", (e) => {
	if (!menuEl.contains(e.target)) menuEl.style.display = "none";
});

render();
</script>
</body>
</html>
```
